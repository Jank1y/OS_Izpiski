🔗Mehke povezave — relativna pot POGOSTO NA KOLOKVIJU

ln -s cilj imeSimbolična (mehka) vez
ln cilj imeTrda vez
readlink imeKam kaže vez
# Si v /home/student, datoteka je /etc/passwd
ln -s ../../etc/passwd geslo

# Si v /home/student, datoteka je /media
ln -s ../../media mount

# Si v /home/student, datoteka je /mod (korenski)

Štej nivoje: /home/student → 2× ../ = prideš v /
⚠ Simbolična vez → kaže na ime (original se ne izbriše)
Trda vez → kaže na inode (podatki ostanejo dokler obstaja vsaj ena vez)

📜Bash skripta — primeri nalog iz kolokvija

#!/bin/bash

# PRIMER 1: Skupna raba imenika (kolokvij 2024)
imenik="${1:-skupnaraba}" # privzeta vrednost
mkdir /home/"$imenik" || { echo "Imenik že obstaja"; exit 5; }
groupadd sodelovanje && echo "Ustvarjena nova skupina za skupno rabo" \
  || echo "Skupina za skupno rabo že obstaja"
chgrp sodelovanje /home/"$imenik"
chmod 1770 /home/"$imenik" # sticky bit

# PRIMER 2: Preveri če si root
if [ $UID -ne 0 ]; then
  echo "Skripto moraš pognati kot root!"; exit 1
fi
last -n 5 # zadnjih 5 uspešnih prijav
lastb -n 5 # zadnjih 5 neuspešnih prijav

# PRIMER 3: Preveri ali user obstaja
if [ -z "$1" ]; then
  echo "Podaj uporabnika kot argument!"; exit 1
fi
grep "^$1:" /etc/passwd && echo "User obstaja" || echo "User ne obstaja"
if [ -d "/home/$1" ]; then
  echo "Domači imenik obstaja"
  ls -l /home/$1
fi

# PRIMER 4: Ustvari userja + dodaj v sudo
sudo useradd -m -c "Miha Krajnc" -d /home/miha -p $(mkpasswd geslo) miha \
  || { echo "Nimaš pravic ali pa uporabnik že obstaja."; exit 42; }
sudo usermod -aG sudo miha
